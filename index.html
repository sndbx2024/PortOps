function doPost(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Sheet1'); // your sheet name var data = JSON.parse(e.postData.contents); var barcode = data.barcode; var values = sheet.getDataRange().getValues(); var headers = values[0]; // first row = headers // Find the column index for "TCN" and "status" by header name var tcnCol = headers.indexOf('TCN'); var statusCol = headers.indexOf('status'); // Safety check: make sure both columns exist if (tcnCol === -1 || statusCol === -1) { return ContentService.createTextOutput( JSON.stringify({ result: 'error', message: 'TCN or status column not found' }) ); } // Search for the matching TCN value for (var i = 1; i < values.length; i++) { // skip header row if (String(values[i][tcnCol]) === String(barcode)) { sheet.getRange(i + 1, statusCol + 1).setValue('loaded'); return ContentService.createTextOutput(JSON.stringify({ result: 'success' })); } } return ContentService.createTextOutput(JSON.stringify({ result: 'not found' })); }
